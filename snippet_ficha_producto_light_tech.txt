// Panel B2B: invitacion a registro debajo del precio (solo si NO es Cliente Comunidad)
add_action('woocommerce_single_product_summary', function() {
    if (is_user_logged_in()) {
        $user = wp_get_current_user();
        if (in_array('community_customer', (array) $user->roles)) return;
    }
    $registro_url = wc_get_page_permalink('myaccount') . '?action=register';
    ?>
    <div class="csm-b2b-panel">
        <div class="label">Precio empresa</div>
        <div class="msg">¿Compras para tu <strong>empresa, colegio o institución</strong>? Regístrate como Cliente Comunidad y accede a precio mayorista en este producto.</div>
        <a href="<?php echo esc_url($registro_url); ?>">Registrarme como empresa →</a>
    </div>
    <?php
}, 25);
